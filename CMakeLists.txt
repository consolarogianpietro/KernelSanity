cmake_minimum_required(VERSION 3.16)
project(KernelSanity VERSION 0.1.0 LANGUAGES CXX)
add_library(kernelsanity INTERFACE)
add_library(ks::kernelsanity ALIAS kernelsanity)
target_include_directories(kernelsanity INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(kernelsanity INTERFACE cxx_std_17)
add_executable(ks_example examples/compare.cpp)
target_link_libraries(ks_example PRIVATE ks::kernelsanity)
add_executable(ks_replay_example examples/replay.cpp)
target_link_libraries(ks_replay_example PRIVATE ks::kernelsanity)
add_executable(ks_boundary_example examples/boundaries.cpp)
target_link_libraries(ks_boundary_example PRIVATE ks::kernelsanity)
add_executable(ks_multi_example examples/multi_compare.cpp)
target_link_libraries(ks_multi_example PRIVATE ks::kernelsanity)
include(CTest)
if(BUILD_TESTING)
  add_executable(ks_tests tests/compare_test.cpp)
  target_link_libraries(ks_tests PRIVATE ks::kernelsanity)
  add_test(NAME kernelsanity COMMAND ks_tests)
  add_test(NAME replay_example COMMAND ks_replay_example)
  add_test(NAME boundary_example COMMAND ks_boundary_example)
  add_executable(ks_multi_tests tests/multi_compare_test.cpp)
  target_link_libraries(ks_multi_tests PRIVATE ks::kernelsanity)
  add_test(NAME multi_input COMMAND ks_multi_tests)
  add_test(NAME multi_example COMMAND ks_multi_example)
endif()
